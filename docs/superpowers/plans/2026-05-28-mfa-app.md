```swift
import CryptoKit
import Foundation

enum TOTPAlgorithm: String, Codable, CaseIterable {
    case sha1 = "SHA1"
    case sha256 = "SHA256"
    case sha512 = "SHA512"

    var displayName: String { rawValue }

    func hmac(key: Data, data: Data) -> Data {
        switch self {
        case .sha1:
            return Data(HMAC<Insecure.SHA1>.authenticationCode(for: data, using: SymmetricKey(data: key)))
        case .sha256:
            return Data(HMAC<SHA256>.authenticationCode(for: data, using: SymmetricKey(data: key)))
        case .sha512:
            return Data(HMAC<SHA512>.authenticationCode(for: data, using: SymmetricKey(data: key)))
        }
    }
}
```

`Sources/MFA/Services/TOTPEngine.swift`:

```swift
import CryptoKit
import Foundation

enum TOTPEngine {

    /// Generate a TOTP code.
    /// - Parameters:
    ///   - secret: The raw secret key bytes (already decoded from base32).
    ///   - algorithm: Hash algorithm (SHA1, SHA256, SHA512).
    ///   - digits: Number of digits in the output code (6 or 8).
    ///   - period: Time step in seconds (default 30).
    ///   - timestamp: Unix timestamp. Defaults to current time.
    /// - Returns: Zero-padded TOTP code string.
    static func generate(
        secret: Data,
        algorithm: TOTPAlgorithm,
        digits: Int,
        period: Int = 30,
        timestamp: Int64 = Int64(Date().timeIntervalSince1970)
    ) -> String {
        let counter = timestamp / Int64(period)
        let counterData = withUnsafeBytes(of: counter.bigEndian) { Data($0) }

        let hmacResult = algorithm.hmac(key: secret, data: counterData)

        // Dynamic truncation (RFC 4226, Section 5.4)
        let offset = Int(hmacResult[hmacResult.count - 1] & 0x0F)
        let truncated = (Int(hmacResult[offset]) & 0x7F) << 24
            | (Int(hmacResult[offset + 1]) & 0xFF) << 16
            | (Int(hmacResult[offset + 2]) & 0xFF) << 8
            | (Int(hmacResult[offset + 3]) & 0xFF)

        let modulus = Int(truncating: NSDecimalNumber(decimal: pow(10, digits)))
        let code = truncated % modulus

        return String(format: "%0\(digits)d", code)
    }

    /// Calculate seconds remaining in the current TOTP period.
    static func remainingSeconds(period: Int = 30, timestamp: Int64 = Int64(Date().timeIntervalSince1970)) -> Int {
        return period - Int(timestamp % Int64(period))
    }
}
```

```swift
import Foundation
import Security

enum KeychainService {
    private static let serviceIdentifier = "com.mfa.app.encryption-key"

    /// Retrieve or create the AES-256 encryption key stored in Keychain.
    static func getOrCreateEncryptionKey() throws -> Data {
        if let existing = queryKeychain() {
            return existing
        }
        let newKey = generateRandomKey()
        try saveToKeychain(newKey)
        return newKey
    }

    private static func generateRandomKey() -> Data {
        let key = SymmetricKey(size: .bits256)
        return key.withUnsafeBytes { Data($0) }
    }

    private static func queryKeychain() -> Data? {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: serviceIdentifier,
            kSecAttrAccount as String: "encryption-key",
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne,
        ]
        var result: AnyObject?
        let status = SecItemCopyMatching(query as CFDictionary, &result)
        guard status == errSecSuccess else { return nil }
        return result as? Data
    }

    private static func saveToKeychain(_ data: Data) throws {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: serviceIdentifier,
            kSecAttrAccount as String: "encryption-key",
            kSecValueData as String: data,
            kSecAttrAccessible as String: kSecAttrAccessibleAfterFirstUnlock,
        ]
        let status = SecItemAdd(query as CFDictionary, nil)
        guard status == errSecSuccess else {
            throw KeychainError.saveFailed(status)
        }
    }

    enum KeychainError: Error {
        case saveFailed(OSStatus)
    }
}
```

```swift
import CryptoKit
import Foundation

enum CryptoService {
    private static let nonceLength = 12

    /// Encrypt data using AES-256-GCM. Returns nonce + ciphertext + tag.
    static func encrypt(_ plaintext: Data) throws -> Data {
        let key = try KeychainService.getOrCreateEncryptionKey()
        let symmetricKey = SymmetricKey(data: key)
        let nonce = AES.GCM.Nonce()
        let sealed = try AES.GCM.seal(plaintext, using: symmetricKey, nonce: nonce)
        // Combine: nonce (12 bytes) + ciphertext + tag (16 bytes)
        return sealed.nonce + sealed.ciphertext + sealed.tag
    }

    /// Decrypt data encrypted by `encrypt(_:)`.
    static func decrypt(_ combined: Data) throws -> Data {
        guard combined.count > nonceLength + 16 else {
            throw CryptoError.invalidData
        }
        let key = try KeychainService.getOrCreateEncryptionKey()
        let symmetricKey = SymmetricKey(data: key)

        let nonce = combined.prefix(nonceLength)
        let tagStart = combined.count - 16
        let ciphertext = combined[nonceLength..<tagStart]
        let tag = combined[tagStart...]

        let sealedBox = try AES.GCM.SealedBox(
            nonce: AES.GCM.Nonce(data: nonce),
            ciphertext: Data(ciphertext),
            tag: Data(tag)
        )
        return try AES.GCM.open(sealedBox, using: symmetricKey)
    }

    enum CryptoError: Error {
        case invalidData
    }
}
```

```swift
        let dbPath: String
        if let path {
            dbPath = path
        } else {
            let appSupport = FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask).first!
            let dir = appSupport.appendingPathComponent("MFA", isDirectory: true)

            dbPath = dir.appendingPathComponent(dbName).path

        config.prepareDatabase { db in
            // WAL mode for concurrent reads
            try db.execute(sql: "PRAGMA journal_mode=WAL")
        }

        let dbPool = try DatabasePool(path: dbPath, configuration: config)
        try migrate(dbPool)
        return dbPool
    }

    static func openInMemory() throws -> DatabasePool {
        let dbPool = try DatabasePool(configuration: Configuration())
        try migrate(dbPool)
        return dbPool

        var migrator = DatabaseMigrator()
```

```swift
@MainActor
final class AccountStore: ObservableObject {
    private let dbPool: DatabasePool

    @Published var accounts: [Account] = []

    init(inMemory: Bool = false) throws {
        self.dbPool = inMemory
            ? try DatabaseManager.openInMemory()
            : try DatabaseManager.openDatabase()
    }

    func close() throws {
        // DatabasePool cleans up on dealloc
    }

    func save(_ account: Account) async throws -> Account {
        var mutable = account
        if mutable.sortOrder == 0 {
            let count = try await fetchAll().count
            mutable.sortOrder = count
        }
        let saved = try dbPool.write { db in
            try mutable.save(db)
            return mutable
        }
        await refresh()
        return saved
    }

    func delete(_ account: Account) async throws {
        _ = try dbPool.write { db in
            try account.delete(db)
        }
        await refresh()
    }

    func fetchAll() async throws -> [Account] {
        try dbPool.read { db in
            try Account.order(Account.Columns.sortOrder).fetchAll(db)
        }
    }

    func search(query: String) async throws -> [Account] {
        try dbPool.read { db in
            try Account
                .filter(
                    Account.Columns.issuer.like("%\(query)%") ||
                    Account.Columns.accountName.like("%\(query)%")
                )
                .order(Account.Columns.sortOrder)
                .fetchAll(db)
        }
    }

    func reorder(_ ids: [UUID]) async throws {
        try dbPool.write { db in
            for (index, id) in ids.enumerated() {
                try db.execute(
                    sql: "UPDATE \(Account.databaseTableName) SET sortOrder = ? WHERE id = ?",
                    arguments: [index, id.uuidString]
                )
            }
        }
        await refresh()
    }

    func refresh() async {
        if let fetched = try? await fetchAll() {
            self.accounts = fetched
        }
    }
}
```

```swift
import Foundation

struct ParsedOTPAuth {
    let issuer: String
    let accountName: String
    let secretBase32: String
    let algorithm: TOTPAlgorithm
    let digits: Int
    let period: Int
}

enum OTPAuthParser {

    /// Parse an `otpauth://totp/` URI into structured data.
    static func parse(_ uriString: String) throws -> ParsedOTPAuth {
        guard let url = URLComponents(string: uriString),
              url.scheme == "otpauth",
              url.host == "totp"
        else {
            throw OTPAuthError.invalidURI("Must be an otpauth://totp/ URI")
        }

        // Parse label: "issuer:accountName" or just "accountName"
        let label = url.path.trimmingCharacters(in: CharacterSet(charactersIn: "/"))
        let decodedLabel = label.removingPercentEncoding ?? label

        let issuer: String
        let accountName: String

        if let colonIndex = decodedLabel.firstIndex(of: ":") {
            issuer = String(decodedLabel[..<colonIndex]).trimmingCharacters(in: .whitespaces)
            accountName = String(decodedLabel[label.index(after: colonIndex)...]).trimmingCharacters(in: .whitespaces)
        } else {
            accountName = decodedLabel
            issuer = ""
        }

        // Parse query parameters
        var params = [String: String]()
        if let queryItems = url.queryItems {
            for item in queryItems {
                params[item.name] = item.value
            }
        }

        guard let secret = params["secret"], !secret.isEmpty else {
            throw OTPAuthError.missingSecret
        }

        let finalIssuer = params["issuer"]?.removingPercentEncoding ?? issuer
        guard !finalIssuer.isEmpty else {
            throw OTPAuthError.missingIssuer
        }

        let algorithm = TOTPAlgorithm(rawValue: params["algorithm"]?.uppercased() ?? "SHA1") ?? .sha1
        let digits = Int(params["digits"] ?? "6") ?? 6
        let period = Int(params["period"] ?? "30") ?? 30

        return ParsedOTPAuth(
            issuer: finalIssuer,
            accountName: accountName,
            secretBase32: secret,
            algorithm: algorithm,
            digits: digits,
            period: period
        )
    }

    enum OTPAuthError: Error {
        case invalidURI(String)
        case missingSecret
        case missingIssuer
    }
}
```

```swift
import Foundation

struct GoogleAuthAccount {
    let name: String
    let secretBase32: String
    let algorithm: TOTPAlgorithm
    let digits: Int
    let type: OTPType

    enum OTPType: Int {
        case unspecified = 0
        case hotp = 1
        case totp = 2
    }
}

enum GoogleAuthParser {

    /// Parse an `otpauth-migration://offline?data=...` URI.
    static func parse(_ uriString: String) throws -> [GoogleAuthAccount] {
        guard let url = URLComponents(string: uriString),
              url.scheme == "otpauth-migration",
              url.host == "offline"
        else {
            throw GoogleAuthError.invalidURI
        }

        guard let dataParam = url.queryItems?.first(where: { $0.name == "data" })?.value else {
            throw GoogleAuthError.missingData
        }

        let decodedData = dataParam.removingPercentEncoding ?? dataParam
        guard let protobufData = Data(base64Encoded: decodedData) else {
            throw GoogleAuthError.invalidBase64
        }

        return try parseProtobuf(protobufData)
    }

    // MARK: - Minimal Protobuf Parser

    private static func parseProtobuf(_ data: Data) throws -> [GoogleAuthAccount] {
        let reader = ProtobufReader(data: data)
        var accounts = [GoogleAuthAccount]()

        while let (fieldNumber, wireType) = reader.readTag() {
            switch (fieldNumber, wireType) {
            case (1, .lengthDelimited):
                let accountData = try reader.readLengthDelimited()
                if let account = try parseAccountValue(accountData) {
                    accounts.append(account)
                }
            case (2, .varint): // version
                _ = try reader.readVarint()
            case (3, .varint): // batchSize
                _ = try reader.readVarint()
            case (4, .varint): // batchIndex
                _ = try reader.readVarint()
            case (5, .varint): // batchId
                _ = try reader.readVarint()
            default:
                try reader.skip(wireType: wireType)
            }
        }

        return accounts
    }

    private static func parseAccountValue(_ data: Data) throws -> GoogleAuthAccount? {
        let reader = ProtobufReader(data: data)

        var name = ""
        var secretData = Data()
        var algorithm: TOTPAlgorithm = .sha1
        var digitCount = 6
        var type: GoogleAuthAccount.OTPType = .totp

        while let (fieldNumber, wireType) = reader.readTag() {
            switch (fieldNumber, wireType) {
            case (1, .lengthDelimited): // name
                name = try reader.readString()
            case (2, .lengthDelimited): // secret (raw bytes)
                secretData = try reader.readLengthDelimited()
            case (3, .varint): // algorithm
                let raw = try reader.readVarint()
                algorithm = switch raw {
                case 1: .sha1
                case 2: .sha256
                case 3: .sha512
                default: .sha1
                }
            case (4, .varint): // digits
                let raw = try reader.readVarint()
                digitCount = (raw == 2) ? 8 : 6
            case (5, .varint): // type
                let raw = try reader.readVarint()
                type = GoogleAuthAccount.OTPType(rawValue: Int(raw)) ?? .totp
            default:
                try reader.skip(wireType: wireType)
            }
        }

        guard !secretData.isEmpty else { return nil }

        // Convert raw secret bytes to base32
        let secretBase32 = base32Encode(secretData)

        return GoogleAuthAccount(
            name: name,
            secretBase32: secretBase32,
            algorithm: algorithm,
            digits: digitCount,
            type: type
        )
    }

    // MARK: - Base32 Encoding

    private static func base32Encode(_ data: Data) -> String {
        let alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"
        var result = ""
        var buffer: UInt64 = 0
        var bits = 0

        for byte in data {
            buffer = (buffer << 8) | UInt64(byte)
            bits += 8
            while bits >= 5 {
                bits -= 5
                let index = Int((buffer >> bits) & 0x1F)
                result.append(alphabet[alphabet.index(alphabet.startIndex, offsetBy: index)])
            }
        }

        if bits > 0 {
            let index = Int((buffer << (5 - bits)) & 0x1F)
            result.append(alphabet[alphabet.index(alphabet.startIndex, offsetBy: index)])
        }

        return result
    }

    enum GoogleAuthError: Error {
        case invalidURI
        case missingData
        case invalidBase64
        case parsingFailed
    }
}

// MARK: - Protobuf Reader

private enum WireType: UInt8 {
    case varint = 0
    case fixed64 = 1
    case lengthDelimited = 2
    case fixed32 = 5
}

private final class ProtobufReader {
    private let data: Data
    private var offset = 0

    init(data: Data) {
        self.data = data
    }

    func readTag() -> (fieldNumber: Int, wireType: WireType)? {
        guard offset < data.count else { return nil }
        guard let tag = readVarintUnchecked() else { return nil }
        let fieldNumber = Int(tag >> 3)
        guard let wt = WireType(rawValue: UInt8(tag & 0x07)) else { return nil }
        return (fieldNumber, wt)
    }

    func readVarint() throws -> UInt64 {
        guard offset < data.count else { throw GoogleAuthParser.GoogleAuthError.parsingFailed }
        return readVarintUnchecked()!
    }

    private func readVarintUnchecked() -> UInt64? {
        var result: UInt64 = 0
        var shift = 0
        while offset < data.count {
            let byte = data[offset]
            offset += 1
            result |= UInt64(byte & 0x7F) << shift
            if byte & 0x80 == 0 { return result }
            shift += 7
            if shift > 63 { return nil }
        }
        return nil
    }

    func readLengthDelimited() throws -> Data {
        let length = try Int(readVarint())
        guard offset + length <= data.count else { throw GoogleAuthParser.GoogleAuthError.parsingFailed }
        let result = data[offset..<(offset + length)]
        offset += length
        return Data(result)
    }

    func readString() throws -> String {
        let data = try readLengthDelimited()
        return String(data: data, encoding: .utf8) ?? ""
    }

    func skip(wireType: WireType) throws {
        switch wireType {
        case .varint:
            _ = try readVarint()
        case .fixed64:
            offset += 8
        case .lengthDelimited:
            let length = try Int(readVarint())
            offset += length
        case .fixed32:
            offset += 4
        }
    }
}
```

`Sources/MFA/Services/Base32.swift`:

```swift
import Foundation

enum Base32 {

    /// Decode a base32-encoded string to raw bytes.
    static func decode(_ string: String) -> Data? {
        let alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"
        var result = [UInt8]()
        var buffer: UInt64 = 0
        var bits = 0

        let cleaned = string.uppercased()
            .replacingOccurrences(of: "=", with: "")
            .replacingOccurrences(of: " ", with: "")
            .replacingOccurrences(of: "-", with: "")

        for char in cleaned {
            guard let value = alphabet.firstIndex(of: char).map({ UInt64(alphabet.distance(from: alphabet.startIndex, to: $0)) }) else {
                continue
            }
            buffer = (buffer << 5) | value
            bits += 5
            if bits >= 8 {
                bits -= 8
                result.append(UInt8(truncatingIfNeeded: buffer >> bits))
            }
        }

        return Data(result)
    }

    /// Encode raw bytes to base32 string (no padding).
    static func encode(_ data: Data) -> String {
        let alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"
        var result = ""
        var buffer: UInt64 = 0
        var bits = 0

        for byte in data {
            buffer = (buffer << 8) | UInt64(byte)
            bits += 8
            while bits >= 5 {
                bits -= 5
                let index = Int((buffer >> bits) & 0x1F)
                result.append(alphabet[alphabet.index(alphabet.startIndex, offsetBy: index)])
            }
        }

        if bits > 0 {
            let index = Int((buffer << (5 - bits)) & 0x1F)
            result.append(alphabet[alphabet.index(alphabet.startIndex, offsetBy: index)])
        }

        return result
    }
}
```

```swift
@MainActor
final class CodeViewModel: ObservableObject {
    let account: Account
    let decryptedSecret: Data

    @Published var code: String = ""
    @Published var remainingSeconds: Int = 0
    @Published var progress: Double = 1.0

    private var timer: Timer?

        self.account = account
        self.decryptedSecret = decryptedSecret
        refreshCode()
        startTimer()
    }

    deinit {
        timer?.invalidate()
    }

    private func startTimer() {
        timer = Timer.scheduledTimer(withTimeInterval: 0.5, repeats: true) { [weak self] _ in
            Task { @MainActor in
                self?.refreshCode()
            }
        }
    }

    private func refreshCode() {
        let now = Int64(Date().timeIntervalSince1970)
        code = TOTPEngine.generate(
            secret: decryptedSecret,
            algorithm: account.algorithm,
            digits: account.digits,
            period: account.period,
            timestamp: now
        )
        remainingSeconds = TOTPEngine.remainingSeconds(period: account.period, timestamp: now)
        progress = Double(remainingSeconds) / Double(account.period)
    }

    func copyCode() {
        NSPasteboard.general.clearContents()
        NSPasteboard.general.setString(code, forType: .string)
    }
```

```swift
import Combine
import Foundation

@MainActor
final class AccountsViewModel: ObservableObject {
    private let store: AccountStore

    @Published var accounts: [Account] = []
    @Published var codeViewModels: [UUID: CodeViewModel] = [:]
    @Published var searchQuery = ""
    @Published var selectedAccountId: UUID?

    var filteredAccounts: [Account] {
        guard !searchQuery.isEmpty else { return accounts }
        return accounts.filter {
            $0.issuer.localizedCaseInsensitiveContains(searchQuery) ||
            $0.accountName.localizedCaseInsensitiveContains(searchQuery)
        }
    }

    init(store: AccountStore) {
        self.store = store
    }

    func loadAccounts() async {
        await store.refresh()
        accounts = store.accounts
        await MainActor.run {
            rebuildCodeViewModels()
        }
    }

    func addAccount(issuer: String, accountName: String, secretBase32: String, algorithm: TOTPAlgorithm = .sha1, digits: Int = 6, period: Int = 30) async throws {
        guard let secretData = Base32.decode(secretBase32) else {
            throw AccountsError.invalidSecret
        }
        let encrypted = try CryptoService.encrypt(secretData)
        let account = Account(
            issuer: issuer,
            accountName: accountName,
            encryptedSecret: encrypted,
            algorithm: algorithm,
            digits: digits,
            period: period
        )
        _ = try await store.save(account)
        await loadAccounts()
    }

    func deleteAccount(_ account: Account) async throws {
        try await store.delete(account)
        await loadAccounts()
    }

    func importFromOTPAuth(_ uri: String) async throws {
        let parsed = try OTPAuthParser.parse(uri)
        try await addAccount(
            issuer: parsed.issuer,
            accountName: parsed.accountName,
            secretBase32: parsed.secretBase32,
            algorithm: parsed.algorithm,
            digits: parsed.digits,
            period: parsed.period
        )
    }

    func importFromGoogleAuth(_ uri: String) async throws -> Int {
        let googleAccounts = try GoogleAuthParser.parse(uri)
        for ga in googleAccounts {
            guard ga.type == .totp else { continue }
            try await addAccount(
                issuer: extractIssuer(from: ga.name),
                accountName: extractAccountName(from: ga.name),
                secretBase32: ga.secretBase32,
                algorithm: ga.algorithm,
                digits: ga.digits
            )
        }
        return googleAccounts.count
    }

    func reorderAccounts(_ ids: [UUID]) async throws {
        try await store.reorder(ids)
        await loadAccounts()
    }

    private func rebuildCodeViewModels() {
        var newViewModels: [UUID: CodeViewModel] = [:]
        for account in accounts {
            if let existing = codeViewModels[account.id] {
                newViewModels[account.id] = existing
            } else {
                do {
                    let secret = try CryptoService.decrypt(account.encryptedSecret)
                    newViewModels[account.id] = CodeViewModel(account: account, decryptedSecret: secret)
                } catch {
                    // Skip accounts we can't decrypt
                }
            }
        }
        codeViewModels = newViewModels
    }

    private func extractIssuer(from name: String) -> String {
        if let colonIndex = name.firstIndex(of: ":") {
            return String(name[..<colonIndex]).trimmingCharacters(in: .whitespaces)
        }
        return name
    }

    private func extractAccountName(from name: String) -> String {
        if let colonIndex = name.firstIndex(of: ":") {
            return String(name[name.index(after: colonIndex)...]).trimmingCharacters(in: .whitespaces)
        }
        return name
    }

    enum AccountsError: Error {
        case invalidSecret
    }
}
```

```swift
@MainActor
final class QRScanService: NSObject, ObservableObject {
    @Published var scannedValue: String?
    @Published var isScanning = false
    @Published var errorMessage: String?

    private var captureSession: AVCaptureSession?
    private var previewLayer: AVCaptureVideoPreviewLayer?

    func startScanning() {
        guard let device = AVCaptureDevice.default(.builtInWideAngleCamera, for: .video, position: .unspecified) else {
            errorMessage = "No camera available"
            return
        }

        guard AVCaptureDevice.authorizationStatus(for: .video) == .authorized else {
            requestCameraPermission()
            return
        }

        setupCaptureSession(device)
    }

    func stopScanning() {
        captureSession?.stopRunning()
        isScanning = false
    }

    private func requestCameraPermission() {
        AVCaptureDevice.requestAccess(for: .video) { [weak self] granted in
            Task { @MainActor in
                if granted, let device = AVCaptureDevice.default(.builtInWideAngleCamera, for: .video, position: .unspecified) {
                    self?.setupCaptureSession(device)
                } else {
                    self?.errorMessage = "Camera permission denied"
                }
            }
        }
    }

    private func setupCaptureSession(_ device: AVCaptureDevice) {
        let session = AVCaptureSession()
        do {
            let input = try AVCaptureDeviceInput(device: device)
            session.addInput(input)
        } catch {
            errorMessage = error.localizedDescription
            return
        }

        let output = AVCaptureVideoDataOutput()
        output.setSampleBufferDelegate(self, queue: DispatchQueue(label: "qr.scan.queue"))
        session.addOutput(output)

        self.captureSession = session
        isScanning = true

        DispatchQueue.global(qos: .userInitiated).async {
            session.startRunning()
        }
    }

    func makePreviewView() -> NSView {
        let view = NSView()
        guard let session = captureSession else { return view }
        let preview = AVCaptureVideoPreviewLayer(session: session)
        preview.videoGravity = .resizeAspectFill
        view.layer = preview
        view.wantsLayer = true
        self.previewLayer = preview
        return view
    }
}

extension QRScanService: AVCaptureVideoDataOutputSampleBufferDelegate {
    nonisolated func captureOutput(_ output: AVCaptureOutput, didOutput sampleBuffer: CMSampleBuffer, from connection: AVCaptureConnection) {
        guard let pixelBuffer = sampleBuffer.imageBuffer else { return }

        let request = VNDetectBarcodesRequest { [weak self] request, _ in
            for result in request.results as? [VNBarcodeObservation] ?? [] {
                guard let payload = result.payloadStringValue else { continue }
                Task { @MainActor in
                    self?.handleScanResult(payload)
                }
            }
        }

        let handler = VNImageRequestHandler(cvPixelBuffer: pixelBuffer, options: [:])
        try? handler.perform([request])
    }

    @MainActor
    private func handleScanResult(_ payload: String) {
        guard scannedValue == nil else { return }
        scannedValue = payload
        stopScanning()
    }
}
```

```swift
import CoreImage.CIFilterBuiltins
import Foundation
import UniformTypeIdentifiers

struct ExportAccount: Codable {
    let issuer: String
    let accountName: String
    let secretBase32: String
    let algorithm: String
    let digits: Int
    let period: Int
}

enum ExportService {

    /// Export accounts to a JSON file.
    static func exportToJSON(_ accounts: [Account]) throws -> URL {
        var exportAccounts: [ExportAccount] = []
        for account in accounts {
            let secret = try CryptoService.decrypt(account.encryptedSecret)
            exportAccounts.append(ExportAccount(
                issuer: account.issuer,
                accountName: account.accountName,
                secretBase32: Base32.encode(secret),
                algorithm: account.algorithm.rawValue,
                digits: account.digits,
                period: account.period
            ))
        }

        let data = try JSONEncoder().encode(exportAccounts)
        let dataWithFormatting = try JSONSerialization.jsonObject(with: data)
        let prettyData = try JSONSerialization.data(withJSONObject: dataWithFormatting, options: [.prettyPrinted, .sortedKeys])

        let panel = NSSavePanel()
        panel.allowedContentTypes = [UTType.json]
        panel.nameFieldStringValue = "mfa-accounts.json"
        panel.canCreateDirectories = true

        guard panel.runModal() == .OK, let url = panel.url else {
            throw ExportError.cancelled
        }

        try prettyData.write(to: url)
        return url
    }

    /// Generate a QR code image for an otpauth:// URI.
    static func generateQRCode(for account: Account) throws -> NSImage {
        let secret = try CryptoService.decrypt(account.encryptedSecret)
        let secretBase32 = Base32.encode(secret)

        let uri = "otpauth://totp/\(account.issuer):\(account.accountName)?secret=\(secretBase32)&issuer=\(account.issuer)&algorithm=\(account.algorithm.rawValue)&digits=\(account.digits)&period=\(account.period)"

        let encodedURI = uri.addingPercentEncoding(withAllowedCharacters: .urlQueryAllowed) ?? uri

        let context = CIContext()
        let filter = CIFilter.qrCodeGenerator()
        filter.message = Data(encodedURI.utf8)
        filter.correctionLevel = "M"

        guard let outputImage = filter.outputImage else {
            throw ExportError.qrGenerationFailed
        }

        let transform = CGAffineTransform(scaleX: 10, y: 10)
        let scaledImage = outputImage.transformed(by: transform)

        let rep = NSCIImageRep(ciImage: scaledImage)
        return NSImage(ciImage: scaledImage, size: rep.size)
    }

    enum ExportError: Error {
        case cancelled
        case qrGenerationFailed
    }
}
```
